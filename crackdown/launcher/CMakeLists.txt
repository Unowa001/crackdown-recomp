cmake_minimum_required(VERSION 3.25)
project(CrackdownLauncher VERSION 0.3.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK_THIRDPARTY "${CMAKE_CURRENT_SOURCE_DIR}/../../sdk/thirdparty" CACHE PATH "SDK dependencies")
option(LAUNCHER_BUILD_UI "Build the SDL launcher" ON)
option(LAUNCHER_FIDELITYFX "Expose CAS/FSR (requires a matching FidelityFX game build)" OFF)
include(CTest)
add_compile_definitions(LAUNCHER_VERSION_W=L"${PROJECT_VERSION}")
find_package(Threads REQUIRED)
add_library(launcher_core core.cpp platform.cpp ../common/toml_edit.cpp "${SDK_THIRDPARTY}/crypto/sha256.cpp")
target_include_directories(launcher_core PUBLIC . "${SDK_THIRDPARTY}/tomlplusplus/include" PRIVATE "${SDK_THIRDPARTY}/crypto")
target_link_libraries(launcher_core PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(launcher_core PUBLIC shell32 dxgi advapi32 ole32 ws2_32)
endif()
add_library(launcher_v2 xzp.cpp steam.cpp update.cpp)
target_link_libraries(launcher_v2 PUBLIC launcher_core)
if(WIN32)
  target_link_libraries(launcher_v2 PRIVATE winhttp)
endif()
if(BUILD_TESTING)
  add_executable(launcher_test_child test_child.cpp)
  target_link_libraries(launcher_test_child PRIVATE launcher_core)
  add_executable(launcher_tests tests.cpp)
  target_link_libraries(launcher_tests PRIVATE launcher_core)
  add_dependencies(launcher_tests launcher_test_child)
  if(WIN32)
    target_link_libraries(launcher_tests PRIVATE shell32 dxgi)
    target_link_libraries(launcher_test_child PRIVATE shell32)
  endif()
  add_test(NAME launcher_core COMMAND launcher_tests)
  add_executable(launcher_v2_tests v2_tests.cpp)
  target_link_libraries(launcher_v2_tests PRIVATE launcher_v2)
  foreach(case xzp_reader vdf_round_trip update_version sha256_verification staged_replacement)
    add_test(NAME launcher_${case} COMMAND launcher_v2_tests ${case})
  endforeach()
endif()
if(LAUNCHER_BUILD_UI)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${SDK_THIRDPARTY}/sdl3" "${CMAKE_CURRENT_BINARY_DIR}/sdl3" EXCLUDE_FROM_ALL)
  set(IMGUI "${SDK_THIRDPARTY}/imgui")
  add_executable(CrackdownLauncher main.cpp visuals.cpp
    "${IMGUI}/imgui.cpp" "${IMGUI}/imgui_draw.cpp" "${IMGUI}/imgui_tables.cpp" "${IMGUI}/imgui_widgets.cpp"
    "${IMGUI}/backends/imgui_impl_sdl3.cpp" "${IMGUI}/backends/imgui_impl_sdlrenderer3.cpp")
  target_include_directories(CrackdownLauncher PRIVATE "${IMGUI}" "${IMGUI}/backends" "${SDK_THIRDPARTY}/stb" "${SDK_THIRDPARTY}/crypto")
  target_link_libraries(CrackdownLauncher PRIVATE launcher_v2 SDL3::SDL3-static)
  target_compile_definitions(CrackdownLauncher PRIVATE LAUNCHER_VERSION="${PROJECT_VERSION}" LAUNCHER_FIDELITYFX=$<BOOL:${LAUNCHER_FIDELITYFX}>)
  file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/assets/files.txt" LAUNCHER_ASSETS)
  foreach(asset IN LISTS LAUNCHER_ASSETS)
    get_filename_component(asset_dir "${asset}" DIRECTORY)
    add_custom_command(TARGET CrackdownLauncher POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:CrackdownLauncher>/assets/${asset_dir}"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/assets/${asset}" "$<TARGET_FILE_DIR:CrackdownLauncher>/assets/${asset}")
  endforeach()
  add_custom_command(TARGET CrackdownLauncher POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:CrackdownLauncher>/LICENSES"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/OFL.txt" "$<TARGET_FILE_DIR:CrackdownLauncher>/LICENSES/Barlow-OFL.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/assets/glyphs/LICENSE.txt" "$<TARGET_FILE_DIR:CrackdownLauncher>/LICENSES/Controller-prompts-CC0.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/licenses/stb.txt" "$<TARGET_FILE_DIR:CrackdownLauncher>/LICENSES/stb.txt")
  if(WIN32)
    set_target_properties(CrackdownLauncher PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_libraries(CrackdownLauncher PRIVATE shell32 dxgi)
  endif()
endif()
