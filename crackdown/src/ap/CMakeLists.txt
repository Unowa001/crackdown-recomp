# Standalone host verification; does not configure the game/SDK or networking.
cmake_minimum_required(VERSION 3.25)
project(CrackdownAPState LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK_THIRDPARTY "${CMAKE_CURRENT_SOURCE_DIR}/../../../sdk/thirdparty" CACHE PATH "SDK dependencies")
include(CTest)
add_library(crackdown_ap_state session_state.cpp ../../common/toml_edit.cpp)
target_include_directories(crackdown_ap_state PUBLIC . PRIVATE "${SDK_THIRDPARTY}/tomlplusplus/include")
if(BUILD_TESTING)
  add_executable(ap_adapter_tests ../../tests/ap_adapter_tests.cpp)
  target_include_directories(ap_adapter_tests PRIVATE .)
  add_test(NAME ap_adapter_provenance COMMAND ap_adapter_tests)
  set(AP_CATALOG "${CMAKE_CURRENT_SOURCE_DIR}/../../ap/catalog.json")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${AP_CATALOG}")
  file(READ "${AP_CATALOG}" _ap_catalog)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ap_catalog_data.h"
      "#pragma once\ninline constexpr char kCatalogJson[] = R\"apcatalog(${_ap_catalog})apcatalog\";\n"
      "inline constexpr unsigned long kCatalogJsonSize = sizeof(kCatalogJson) - 1;\n")
  add_executable(ap_catalog_tests catalog.cpp ../../tests/ap_catalog_tests.cpp)
  target_include_directories(ap_catalog_tests PRIVATE . "${CMAKE_CURRENT_BINARY_DIR}"
      "${SDK_THIRDPARTY}/inja/third_party/include")
  add_test(NAME ap_catalog_contract COMMAND ap_catalog_tests "${AP_CATALOG}")
  add_executable(ap_state_tests ../../tests/ap_state_tests.cpp)
  target_link_libraries(ap_state_tests PRIVATE crackdown_ap_state)
  foreach(case binding receipts recovery checks persistence lineage acknowledge skills weapons goal)
    add_test(NAME ap_state_${case} COMMAND ap_state_tests ${case})
  endforeach()
endif()
